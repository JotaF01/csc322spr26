// Implicit equation
//  F(x,y) = (x^2+y^2)^2 - 4(x^2-y^2) = 0
//

function F(x, y) { 
  const a = x * x + y * y;     
  const b = x * x - y * y;     
  return (a * a) - 4 * b;      
}

const implicitMaterial = new THREE.LineBasicMaterial({ color: 0xffffff });
const implicitPoints = [];
const N = 720; 

for (let i = 0; i <= N; i++) {
  const theta = (i / N) * Math.PI * 2;      
  const c = Math.cos(2 * theta);            

  if (c < 0) continue;                      

  const r = 2 * Math.sqrt(c);               

  const x = r * Math.cos(theta);            
  const y = r * Math.sin(theta);            

  implicitPoints.push(new THREE.Vector3(x, y, 0));
}

const implicitGeometry = new THREE.BufferGeometry().setFromPoints(implicitPoints);
const implicitCurve = new THREE.Line(implicitGeometry, implicitMaterial);
scene.add(implicitCurve);

const originDot = new THREE.Mesh(
  new THREE.SphereGeometry(0.15, 16, 16),
  new THREE.MeshBasicMaterial({ color: 0xffffff })
);
scene.add(originDot);

// Render Loop
function animate() {
  requestAnimationFrame(animate);
  controls.update();
  renderer.render(scene, camera);
}
